#------------------------------------------------------------------------------
# ports/Darwin -- Nanosaur 2 macOS screen saver (.saver bundle).
#
# Renders the game's attract-mode scene (the level-intro wormhole with
# "PRESS ANY KEY", DoLevelIntroScreen(INTRO_MODE_SCREENSAVER)) as a native
# macOS screen saver. Standalone project, same spirit as ports/3DS: a
# CURATED subset of the engine (the ~40 files the wormhole scene actually
# needs, out of the whole game), no SDL (Saver/shim.c supplies the few
# runtime calls the subset reaches; the real SDL3 headers are used for
# types only), and only the handful of assets the scene loads (~3 MB out
# of the 113 MB Data folder).
#
# Generate the Xcode project with:
#   cmake -G Xcode -B build-xcode ports/Darwin
# or build directly:
#   cmake -B ports/Darwin/build ports/Darwin && cmake --build ports/Darwin/build
#
# Install: copy the built Nanosaur2.saver to ~/Library/Screen Savers/
#
# Architecture notes:
# - Two Swift modules, mirroring the desktop build's MetalRenderer split:
#   the engine subset compiles with the game's bridging header (SwMacTypes
#   collides with AppKit's MacTypes), while the ScreenSaverView host
#   (Nanosaur2SaverView.swift) imports ScreenSaver/AppKit in its own module
#   and reaches the engine through 4 C entry points (SaverGlue.swift).
# - The host view owns the NSOpenGLContext; the engine's
#   NANOSAUR_SCREENSAVER seams (RenderBackend.swift, OGL_Support.swift)
#   skip SDL context/present/window-size handling accordingly.
#------------------------------------------------------------------------------

cmake_minimum_required(VERSION 3.26)

set(GAME_VERSION 2.1.1)

project(Nanosaur2ScreenSaver
	VERSION ${GAME_VERSION}
	LANGUAGES C Swift)

set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version")
if(NOT CMAKE_OSX_ARCHITECTURES)
	set(CMAKE_OSX_ARCHITECTURES "arm64")	# match the desktop Swift build (single-arch)
endif()

set(ENGINE_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(SRC ${ENGINE_ROOT}/Source)
set(SAVER ${CMAKE_CURRENT_SOURCE_DIR}/Saver)
set(DATA ${ENGINE_ROOT}/Data)

#------------------------------------------------------------------------------
# version.h (normally generated by the top-level CMakeLists)
#------------------------------------------------------------------------------

set(GAME_FULL_NAME "Nanosaur 2")
set(GAME_IDENTIFIER "io.jor.nanosaur2.saver")
set(GAME_COPYRIGHT "© 2004 Pangea Software, Inc., © 2025 Iliyas Jorio.")
configure_file(${SRC}/Headers/version.h.in ${SRC}/Headers/version.h)

#------------------------------------------------------------------------------
# METALRENDERER (separate Swift module, no bridging header - same split as
# the desktop CMakeLists: `import Metal` collides with SwMacTypes.h if
# compiled with the game's bridging header)
#------------------------------------------------------------------------------

file(GLOB METALRENDERER_SWIFT ${SRC}/Metal/*.swift)
add_library(MetalRenderer STATIC ${METALRENDERER_SWIFT})
set_target_properties(MetalRenderer PROPERTIES
	Swift_COMPILATION_MODE wholemodule
	XCODE_ATTRIBUTE_SWIFT_OPTIMIZATION_LEVEL[variant=Debug] "-Onone"
)
target_link_libraries(MetalRenderer PUBLIC "-framework Metal" "-framework QuartzCore")

#------------------------------------------------------------------------------
# ENGINE SUBSET (static lib, one Swift module + the leftover C files)
#------------------------------------------------------------------------------

set(ENGINE_SWIFT
	# 3D / rendering
	${SRC}/3D/3DMath_Angles.swift
	${SRC}/3D/3DMath_Geometry.swift
	${SRC}/3D/3DMath_Matrix.swift
	${SRC}/3D/Atlas.swift
	${SRC}/3D/Bg3d.swift
	${SRC}/3D/Camera.swift
	${SRC}/3D/MOVertexArrayDataExtensions.swift
	${SRC}/3D/MetaObjects.swift
	${SRC}/3D/MetaObjectsExtensions.swift
	${SRC}/3D/OGLSetupOutputTypeExtensions.swift
	${SRC}/3D/OGLTypes.swift
	${SRC}/3D/OGL_Support.swift
	${SRC}/3D/QuadMesh.swift
	${SRC}/3D/RenderBackend.swift
	${SRC}/3D/MetalRenderBackend.swift
	${SRC}/3D/Sprites.swift

	# effects the scene uses (jetpack glints, particles infra)
	${SRC}/Effects/Particles.swift
	${SRC}/Effects/Sparkle.swift

	# skeletal animation (wormhole + nano)
	${SRC}/Skeleton/Bones.swift
	${SRC}/Skeleton/SkeletonAnim.swift
	${SRC}/Skeleton/SkeletonExtensions.swift
	${SRC}/Skeleton/SkeletonJoints.swift
	${SRC}/Skeleton/SkeletonObj.swift

	# the scene itself + the 2D overlay state its text/sprites draw with
	${SRC}/Screens/LevelIntroScene.swift
	${SRC}/Screens/Infobar.swift

	# system core
	${SRC}/System/FourCC.swift
	${SRC}/System/GameEngine.swift
	${SRC}/System/GameEnums.swift
	${SRC}/System/Misc.swift
	${SRC}/System/MobjTypes.swift
	${SRC}/System/ObjNodeExtensions.swift
	${SRC}/System/ObjNodeList.swift
	${SRC}/System/Objects.swift
	${SRC}/System/Objects2.swift
	${SRC}/System/SObjTypes.swift
	${SRC}/System/SwiftEnumExtensions.swift
	${SRC}/System/SwiftSupport.swift
	${SRC}/System/File.swift
	${SRC}/System/FileSystem.swift
	${SRC}/System/Localization.swift
	${SRC}/System/Window.swift
	${SRC}/System/UIEffects.swift

	# saver-specific: engine entry points + subsystem shells
	${SAVER}/SaverGlue.swift
	${SAVER}/SaverStubs.swift
)

# BG3DFile/ResourceFile: standalone Swift packages flat-compiled into the
# module, same treatment as the desktop CMakeLists (their BinaryParsing
# imports are canImport-guarded and skipped here).
file(GLOB_RECURSE BG3DFILE_SWIFT ${ENGINE_ROOT}/Sources/BG3DFile/*.swift)
file(GLOB_RECURSE RESOURCEFILE_SWIFT ${ENGINE_ROOT}/Sources/ResourceFile/*.swift)
file(GLOB_RECURSE BINARYPARSING_SWIFT "${ENGINE_ROOT}/extern/swift-binary-parsing/BinaryParsing/*.swift")

set(ENGINE_C
	${SRC}/System/BootGlobals.c
	${SRC}/System/Misc.c
	${SRC}/System/Localization.c
	${SRC}/System/InputDefaults.c
	${SRC}/System/Main.c
	${SRC}/System/stb_image.c
	${SAVER}/shim.c
)

add_library(SaverEngine STATIC
	${ENGINE_SWIFT}
	${BG3DFILE_SWIFT}
	${RESOURCEFILE_SWIFT}
	${BINARYPARSING_SWIFT}
	${ENGINE_C}
)

set_target_properties(SaverEngine PROPERTIES
	Swift_COMPILATION_MODE wholemodule
	# Xcode's canvas preview refuses to load a module unless it was built
	# -Onone ("Cannot preview in this file - Not built with -Onone") - the
	# CMake Xcode generator doesn't set this per-config on its own, unlike
	# an Xcode-templated target, which defaults Debug to -Onone.
	XCODE_ATTRIBUTE_SWIFT_OPTIMIZATION_LEVEL[variant=Debug] "-Onone"
)

target_include_directories(SaverEngine PRIVATE ${SRC}/Headers ${ENGINE_ROOT}/extern/minimp3)
target_compile_definitions(SaverEngine PRIVATE NANOSAUR_SCREENSAVER GL_SILENCE_DEPRECATION)

# SDL3 headers (types only - never linked; see Saver/shim.c)
target_compile_options(SaverEngine PRIVATE
	"$<$<COMPILE_LANGUAGE:C>:-F${ENGINE_ROOT}/extern>"
	$<$<COMPILE_LANGUAGE:C>:-Wno-multichar>
)

target_compile_options(SaverEngine PRIVATE
	$<$<COMPILE_LANGUAGE:Swift>:-import-objc-header>
	$<$<COMPILE_LANGUAGE:Swift>:${SAVER}/SaverBridging.h>
	$<$<COMPILE_LANGUAGE:Swift>:-parse-as-library>
	$<$<COMPILE_LANGUAGE:Swift>:-DNANOSAUR_SCREENSAVER>
	# Required by vendored swift-binary-parsing (see desktop CMakeLists)
	$<$<COMPILE_LANGUAGE:Swift>:-enable-experimental-feature>
	$<$<COMPILE_LANGUAGE:Swift>:Lifetimes>
	# ClangImporter flags: engine headers + SDL3 framework headers + the
	# NANOSAUR_SCREENSAVER define for C code seen from Swift
	$<$<COMPILE_LANGUAGE:Swift>:SHELL:-Xcc -I${SRC}/Headers>
	$<$<COMPILE_LANGUAGE:Swift>:SHELL:-Xcc -I${ENGINE_ROOT}/extern/minimp3>
	$<$<COMPILE_LANGUAGE:Swift>:SHELL:-Xcc -F${ENGINE_ROOT}/extern>
	$<$<COMPILE_LANGUAGE:Swift>:SHELL:-Xcc -DNANOSAUR_SCREENSAVER>
	$<$<COMPILE_LANGUAGE:Swift>:SHELL:-Xcc -DGL_SILENCE_DEPRECATION>
)

target_link_libraries(SaverEngine PUBLIC
	MetalRenderer
	# OpenGL stays linked for GLRenderBackend (compiled as GameEngine's
	# initial renderer; swapped to MetalRenderBackend during boot).
	"-framework OpenGL"
	"-framework Foundation"
)

#------------------------------------------------------------------------------
# THE .saver BUNDLE (ScreenSaverView host, own Swift module)
#------------------------------------------------------------------------------

add_library(Nanosaur2Saver MODULE
	${SAVER}/Nanosaur2SaverView.swift
	${SAVER}/Nanosaur2SaverView+Preview.swift
)

set_target_properties(Nanosaur2Saver PROPERTIES
	OUTPUT_NAME "Nanosaur2"
	BUNDLE TRUE
	BUNDLE_EXTENSION "saver"
	MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in
	MACOSX_BUNDLE_EXECUTABLE_NAME "Nanosaur2"
	MACOSX_BUNDLE_GUI_IDENTIFIER "${GAME_IDENTIFIER}"
	MACOSX_BUNDLE_BUNDLE_NAME "Nanosaur 2"
	MACOSX_BUNDLE_SHORT_VERSION_STRING "${GAME_VERSION}"
	MACOSX_BUNDLE_BUNDLE_VERSION "${GAME_VERSION}"
	MACOSX_BUNDLE_COPYRIGHT "${GAME_COPYRIGHT}"
	Swift_COMPILATION_MODE wholemodule
	# See SaverEngine's matching property above - the previewed target
	# itself needs this too, not just its static-library dependency.
	XCODE_ATTRIBUTE_SWIFT_OPTIMIZATION_LEVEL[variant=Debug] "-Onone"
)

target_compile_options(Nanosaur2Saver PRIVATE
	$<$<COMPILE_LANGUAGE:Swift>:-import-objc-header>
	$<$<COMPILE_LANGUAGE:Swift>:${SAVER}/saver_api.h>
	$<$<COMPILE_LANGUAGE:Swift>:-parse-as-library>
	# Gates Nanosaur2SaverView+Preview.swift's #Preview - Debug only, same
	# as a normal Xcode-templated target's SWIFT_ACTIVE_COMPILATION_CONDITIONS.
	$<$<AND:$<COMPILE_LANGUAGE:Swift>,$<CONFIG:Debug>>:-DDEBUG>
)

target_link_libraries(Nanosaur2Saver PRIVATE
	SaverEngine
	"-framework ScreenSaver"
	"-framework AppKit"
	"-framework OpenGL"
)

#------------------------------------------------------------------------------

# Nanosaur2Saver_Boot). Mirrors ports/3DS/romfs-poc's asset list.

#------------------------------------------------------------------------------
# HEADLESS SMOKE TEST (optional) - offscreen CGL + FBO, same C entry points
# as the .saver; dumps a rendered frame to a PPM for eyeballing.
#------------------------------------------------------------------------------

# main.swift, not SmokeTest.swift: Swift only allows top-level statements
# in a file named main.swift.
add_executable(SaverSmoke ${CMAKE_CURRENT_SOURCE_DIR}/test/main.swift)

target_compile_options(SaverSmoke PRIVATE
	$<$<COMPILE_LANGUAGE:Swift>:-import-objc-header>
	$<$<COMPILE_LANGUAGE:Swift>:${SAVER}/saver_api.h>
)

target_link_libraries(SaverSmoke PRIVATE
	SaverEngine
	"-framework OpenGL"
)

#------------------------------------------------------------------------------

# Explicit POST_BUILD copies (MACOSX_PACKAGE_LOCATION on a MODULE bundle
# only creates the directories under the Ninja generator).
set(BUNDLE_RES "$<TARGET_BUNDLE_DIR:Nanosaur2Saver>/Contents/Resources/Data")

add_custom_command(TARGET Nanosaur2Saver POST_BUILD
	COMMAND ${CMAKE_COMMAND} -E rm -f "$<TARGET_FILE_DIR:Nanosaur2Saver>/Nanosaur2Saver.emit-module.d"
	COMMAND ${CMAKE_COMMAND} -E make_directory
		"${BUNDLE_RES}/Models" "${BUNDLE_RES}/Skeletons" "${BUNDLE_RES}/System"
		"${BUNDLE_RES}/Sprites/textures" "${BUNDLE_RES}/Sprites/menu" "${BUNDLE_RES}/Sprites/fonts"
	COMMAND ${CMAKE_COMMAND} -E copy_if_different
		${DATA}/Models/levelintro.bg3d ${DATA}/Models/playerparts.bg3d
		"${BUNDLE_RES}/Models"
	COMMAND ${CMAKE_COMMAND} -E copy_if_different
		${DATA}/Skeletons/bonusworm.bg3d ${DATA}/Skeletons/bonusworm.skeleton.rsrc
		${DATA}/Skeletons/nano.bg3d ${DATA}/Skeletons/nano.skeleton.rsrc
		"${BUNDLE_RES}/Skeletons"
	COMMAND ${CMAKE_COMMAND} -E copy_if_different
		${DATA}/Sprites/textures/stardome.jpg "${BUNDLE_RES}/Sprites/textures"
	COMMAND ${CMAKE_COMMAND} -E copy_if_different
		${DATA}/Sprites/menu/nanologo.png ${DATA}/Sprites/menu/nanologo.jpg
		"${BUNDLE_RES}/Sprites/menu"
	COMMAND ${CMAKE_COMMAND} -E copy_if_different
		${DATA}/Sprites/fonts/font.png ${DATA}/Sprites/fonts/font.txt
		${DATA}/Sprites/fonts/font.kerning.txt
		"${BUNDLE_RES}/Sprites/fonts"
	COMMAND ${CMAKE_COMMAND} -E copy_directory
		${DATA}/Sprites/particle "${BUNDLE_RES}/Sprites/particle"
	COMMAND ${CMAKE_COMMAND} -E copy_if_different
		${DATA}/System/strings.csv "${BUNDLE_RES}/System"
	# Ad-hoc sign so legacyScreenSaver will load a locally built bundle.
	COMMAND codesign --force --sign - "$<TARGET_BUNDLE_DIR:Nanosaur2Saver>"
	COMMENT "Copying the wormhole scene's assets into Nanosaur2.saver + ad-hoc signing"
	VERBATIM)
